     3→// =============================================================================
     4→// Liturgical calendar — heuristic, deterministic computation of DayInfo.
     5→// All fixed-date feasts are taken from the General Roman Calendar; movable
     6→// feasts are computed via the Computus (Meeus/Jones/Butcher algorithm).
     7→// =============================================================================
     8→
     9→function easterDate(year: number): Date {
    10→  const a = year % 19
    11→  const b = Math.floor(year / 100)
    12→  const c = year % 100
    13→  const d = Math.floor(b / 4)
    14→  const e = b % 4
    15→  const f = Math.floor((b + 8) / 25)
    16→  const g = Math.floor((b - f + 1) / 3)
    17→  const h = (19 * a + b - d - g + 15) % 30
    18→  const i = Math.floor(c / 4)
    19→  const k = c % 4
    20→  const l = (32 + 2 * e + 2 * i - h - k) % 7
    21→  const m = Math.floor((a + 11 * h + 22 * l) / 451)
    22→  const month = Math.floor((h + l - 7 * m + 114) / 31)
    23→  const day = ((h + l - 7 * m + 114) % 31) + 1
    24→  return new Date(year, month - 1, day)
    25→}
    26→
    27→function addDays(d: Date, n: number): Date {
    28→  const r = new Date(d)
    29→  r.setDate(r.getDate() + n)
    30→  return r
    31→}
    32→
    33→function daysBetween(a: Date, b: Date): number {
    34→  const ms = b.getTime() - a.getTime()
    35→  return Math.round(ms / (24 * 60 * 60 * 1000))
    36→}
    37→
    38→function ashWednesday(year: number): Date {
    39→  return addDays(easterDate(year), -46)
    40→}
    41→
    42→function palmSunday(year: number): Date {
    43→  return addDays(easterDate(year), -7)
    44→}
    45→
    46→function pentecostDate(year: number): Date {
    47→  return addDays(easterDate(year), 49)
    48→}
    49→
    50→function trinitySunday(year: number): Date {
    51→  return addDays(pentecostDate(year), 7)
    52→}
    53→
    54→function firstSundayOfAdvent(year: number): Date {
    55→  // 4th Sunday before Christmas (Dec 25). Find the Sunday on or before Dec 25
    56→  // and subtract 3 weeks (21 days).
    57→  const christmas = new Date(year, 11, 25)
    58→  const dow = christmas.getDay()
    59→  const sundayOnOrBefore = addDays(christmas, -dow)
    60→  return addDays(sundayOnOrBefore, -21)
    61→}
    62→
    63→function epiphanySunday(year: number): Date {
    64→  // In France, Epiphany is the Sunday between Jan 2 and Jan 8.
    65→  const jan1 = new Date(year, 0, 1)
    66→  const dow = jan1.getDay()
    67→  const offset = dow === 0 ? 7 : 7 - dow
    68→  return addDays(jan1, offset)
    69→}
    70→
    71→function baptismOfLord(year: number): Date {
    72→  // Sunday after Epiphany Sunday.
    73→  return addDays(epiphanySunday(year), 7)
    74→}

    76→function liturgicalYearLabel(date: Date): 'A' | 'B' | 'C' {
    77→  // Year A starts First Sunday of Advent of years 2016, 2019, 2022, 2025, ...
    78→  // i.e. (year_of_Advent % 3) === 0 -> A, === 1 -> B, === 2 -> C.
    79→  let Y = date.getFullYear()
    80→  let adv = firstSundayOfAdvent(Y)
    81→  if (date.getTime() < adv.getTime()) Y -= 1
    82→  const mod = ((Y % 3) + 3) % 3
    83→  return mod === 0 ? 'A' : mod === 1 ? 'B' : 'C'
    84→}

   202→function findFixedFeast(date: Date): FixedFeast | undefined {
   203→  const m = date.getMonth() + 1
   204→  const d = date.getDate()
   205→  return fixedFeasts.find((f) => f.month === m && f.day === d)
   206→}

   215→function psalterWeekFor(date: Date): string {
   216→  // Deterministic 4-week cycle based on the day of year.
   217→  const startOfYear = new Date(date.getFullYear(), 0, 1)
   218→  const dayOfYear = Math.floor((date.getTime() - startOfYear.getTime()) / (24 * 60 * 60 * 1000))
   219→  const idx = (Math.floor(dayOfYear / 7) + 3) % 4
   220→  const labels = ['I', 'II', 'III', 'IV']
   221→  return `Semaine ${labels[idx]} du Psautier`
   222→}

   224→interface SeasonInfo {
   225→  time: string
   226→  color: LiturgicalColor
   227→  /** Number of the week within this season (for Sunday numbering), 0 if N/A. */
   228→  weekNumber: number
   229→  /** Special override for the day name. */
   230→  dayName?: string
   231→}

   233→function determineSeason(date: Date): SeasonInfo {
   234→  const year = date.getFullYear()
   235→  const easter = easterDate(year)
   236→  const ashWed = ashWednesday(year)
   237→  const holyThu = addDays(easter, -3)
   238→  const goodFri = addDays(easter, -2)
   239→  const holySat = addDays(easter, -1)
   240→  const pentecost = pentecostDate(year)
   241→  const trinity = trinitySunday(year)
   242→  const advent = firstSundayOfAdvent(year)
   243→  const baptism = baptismOfLord(year)
   244→  const christmas = new Date(year, 11, 25)
   245→  const prevChristmas = new Date(year - 1, 11, 25)
   246→  const prevBaptism = baptismOfLord(year - 1)
   247→
   248→  // 1. Christmas season: Dec 25 (prev year) -> Baptism of Lord (this year),
   249→  //    and Dec 25 (this year) -> Baptism of Lord (next year).
   250→  const baptismThisYear = baptism
   251→  const baptismNextYear = baptismOfLord(year + 1)
   252→  if ((date >= prevChristmas && date <= baptismThisYear) || (date >= christmas && date <= baptismNextYear)) {
   253→    // Determine day name within Christmas season.
   254→    if (sameDay(date, christmas)) return { time: 'Temps de Noël', color: 'white', weekNumber: 0, dayName: 'Nativité du Seigneur' }
   255→    if (sameDay(date, prevChristmas)) return { time: 'Temps de Noël', color: 'white', weekNumber: 0, dayName: 'Nativité du Seigneur' }
   256→    if (sameDay(date, baptism) || sameDay(date, prevBaptism) || sameDay(date, baptismThisYear) || sameDay(date, baptismNextYear)) return { time: 'Temps de Noël', color: 'white', weekNumber: 0, dayName: 'Baptême du Seigneur' }
   257→    // Sunday within Octave of Christmas (Holy Family in modern French calendar = Sunday between Dec 25 and Jan 1 inclusive, otherwise Dec 30)
   258→    return { time: 'Temps de Noël', color: 'white', weekNumber: 0, dayName: christmasNameFor(date) }
   259→  }
   260→
   261→  // 2. Ordinary Time, first part: Monday after Baptism -> day before Ash Wednesday.
   262→  if (date > baptism && date < ashWed) {
   263→    const weeksAfterBaptism = Math.floor(daysBetween(baptism, date) / 7)
   264→    const isSunday = date.getDay() === 0
   265→    const n = weeksAfterBaptism // baptism = 0; 1st Sunday = 1
   266→    if (isSunday) {
   267→      return { time: 'Temps ordinaire', color: 'green', weekNumber: n, dayName: `${ordinal(n)} dimanche du Temps ordinaire` }
   268→    }
   269→    return { time: 'Temps ordinaire', color: 'green', weekNumber: n, dayName: weekdayName(date, `${ordinal(n)} semaine du Temps ordinaire`) }
   270→  }
   271→
   272→  // 3. Lent: Ash Wednesday -> Holy Saturday (Easter eve).
   273→  if (date >= ashWed && date < easter) {
   274→    if (sameDay(date, ashWed)) return { time: 'Temps du Carême', color: 'violet', weekNumber: 0, dayName: 'Mercredi des Cendres' }
   275→    if (sameDay(date, palmSunday(year))) return { time: 'Temps du Carême', color: 'red', weekNumber: 6, dayName: 'Dimanche des Rameaux et de la Passion du Seigneur' }
   276→    if (sameDay(date, holyThu)) return { time: 'Triduum pascal', color: 'white', weekNumber: 0, dayName: 'Jeudi Saint' }
   277→    if (sameDay(date, goodFri)) return { time: 'Triduum pascal', color: 'red', weekNumber: 0, dayName: 'Vendredi Saint' }
   278→    if (sameDay(date, holySat)) return { time: 'Triduum pascal', color: 'violet', weekNumber: 0, dayName: 'Samedi Saint' }
   279→
   280→    // Within Lent proper.
   281→    const firstSunOfLent = addDays(ashWed, 7 - ashWed.getDay() + (ashWed.getDay() === 0 ? 0 : 0))
   282→    // Recompute: 1st Sunday of Lent is the Sunday following Ash Wednesday.
   283→    const sun1 = addDays(ashWed, 7 - ashWed.getDay())
   284→    if (date < sun1) {
   285→      // Days after Ash Wed but before 1st Sunday of Lent.
   286→      const isSunday = date.getDay() === 0
   287→      if (isSunday) return { time: 'Temps du Carême', color: 'violet', weekNumber: 1, dayName: '1er dimanche de Carême' }
   288→      return { time: 'Temps du Carême', color: 'violet', weekNumber: 0, dayName: weekdayName(date, 'semaine des Cendres') }
   289→    }
   290→    // Otherwise count Sundays since the 1st Sunday of Lent.
   291→    const weeksAfterSun1 = Math.floor(daysBetween(sun1, date) / 7)
   292→    const n = weeksAfterSun1 + 1
   293→    const isSunday = date.getDay() === 0
   294→    if (isSunday) {
   295→      const color: LiturgicalColor = n === 4 ? 'rose' : 'violet'
   296→      const name = n === 4 ? `4e dimanche de Carême (Laetare)` : `${ordinal(n)} dimanche de Carême`
   297→      return { time: 'Temps du Carême', color, weekNumber: n, dayName: name }
   298→    }
   299→    // Weekdays of Lent — use the week number based on which Sunday’s week we’re in.
   300→    const weekN = n // n is the upcoming Sunday’s number
   301→    return { time: 'Temps du Carême', color: 'violet', weekNumber: weekN, dayName: weekdayName(date, `${ordinal(weekN)} semaine de Carême`) }
   302→  }
   303→
   304→  // 4. Easter season: Easter Sunday -> Pentecost.
   305→  if (date >= easter && date <= pentecost) {
   306→    if (sameDay(date, easter)) return { time: 'Temps pascal', color: 'white', weekNumber: 0, dayName: 'Dimanche de Pâques' }
   307→    if (sameDay(date, addDays(easter, 39))) return { time: 'Temps pascal', color: 'white', weekNumber: 0, dayName: 'Ascension du Seigneur' }
   308→    if (sameDay(date, pentecost)) return { time: 'Temps pascal', color: 'red', weekNumber: 0, dayName: 'Dimanche de la Pentecôte' }
   309→    // Determine week of Easter.
   310→    const daysAfterEaster = daysBetween(easter, date)
   311→    const weekN = Math.floor(daysAfterEaster / 7) + 1
   312→    const isSunday = date.getDay() === 0
   313→    if (isSunday) {
   314→      return { time: 'Temps pascal', color: 'white', weekNumber: weekN, dayName: `${ordinal(weekN)} dimanche de Pâques` }
   315→    }
   316→    // Within the Octave of Easter (first 7 days after Easter).
   317→    if (daysAfterEaster < 7) {
   318→      return { time: 'Temps pascal', color: 'white', weekNumber: 1, dayName: weekdayName(date, "l’Octave de Pâques") }
   319→    }
   320→    return { time: 'Temps pascal', color: 'white', weekNumber: weekN, dayName: weekdayName(date, `${ordinal(weekN)} semaine de Pâques`) }
   321→  }
   322→
   323→  // 5. Trinity Sunday (Sunday after Pentecost).
   324→  if (sameDay(date, trinity)) {
   325→    return { time: 'Temps ordinaire', color: 'white', weekNumber: 0, dayName: 'Sainte Trinité' }
   326→  }
   327→
   328→  // 6. Ordinary Time, second part: Monday after Trinity -> Saturday before Advent Sunday.
   329→  if (date > trinity && date < advent) {
   330→    // Number of Sundays before Lent this year (for continuous numbering).
   331→    const baptismThisYear = baptismOfLord(year)
   332→    const ashWedThisYear = ashWednesday(year)
   333→    let preLentSundays = 0
   334→    let cursor = addDays(baptismThisYear, 7)
   335→    while (cursor < ashWedThisYear) {
   336→      preLentSundays++
   337→      cursor = addDays(cursor, 7)
   338→    }
   339→    const weeksAfterTrinity = Math.floor(daysBetween(trinity, date) / 7)
   340→    const isSunday = date.getDay() === 0
   341→    // Trinity Sunday is celebrated as a separate solemnity; the Sunday after
   342→    // Trinity is the (preLentSundays + 1)th Sunday of Ordinary Time.
   343→    const n = preLentSundays + weeksAfterTrinity
   344→    if (isSunday) {
   345→      // Christ the King (Sunday before Advent) — solemnity.
   346→      if (sameDay(date, addDays(advent, -7))) {
   347→        return { time: 'Temps ordinaire', color: 'white', weekNumber: n, dayName: 'Christ Roi de l\'univers' }
   348→      }
   349→      return { time: 'Temps ordinaire', color: 'green', weekNumber: n, dayName: `${ordinal(n)} dimanche du Temps ordinaire` }
   350→    }
   351→    return { time: 'Temps ordinaire', color: 'green', weekNumber: n, dayName: weekdayName(date, `${ordinal(n)} semaine du Temps ordinaire`) }
   352→  }
   353→
   354→  // 7. Advent: Advent Sunday -> Dec 24.
   355→  if (date >= advent && date < christmas) {
   356→    const weeksAfterAdvent = Math.floor(daysBetween(advent, date) / 7)
   357→    const n = weeksAfterAdvent + 1
   358→    const isSunday = date.getDay() === 0
   359→    if (isSunday) {
   360→      const color: LiturgicalColor = n === 3 ? 'rose' : 'violet'
   361→      const name = n === 3 ? `3e dimanche de l’Avent (Gaudete)` : `${ordinal(n)} dimanche de l’Avent`
   362→      return { time: 'Temps de l\'Avent', color, weekNumber: n, dayName: name }
   363→    }
   364→    return { time: 'Temps de l\'Avent', color: 'violet', weekNumber: n, dayName: weekdayName(date, `${ordinal(n)} semaine de l’Avent`) }
   365→  }
   366→
   367→  // Fallback (shouldn’t happen).
   368→  return { time: 'Temps ordinaire', color: 'green', weekNumber: 0 }
   369→}

   371→function sameDay(a: Date, b: Date): boolean {
   372→  return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate()
   373→}

   375→function weekdayName(date: Date, weekLabel: string): string {
   376→  const wd = WEEKDAY_NAMES[date.getDay()]
   377→  return `${wd} de la ${weekLabel}`
   378→}

   380→function christmasNameFor(date: Date): string {
   381→  // Sunday within the Octave of Christmas — Holy Family.
   382→  if (date.getDay() === 0) return 'Sainte Famille de Jésus, Marie et Joseph'
   383→  const d = date.getDate()
   384→  const m = date.getMonth() + 1
   385→  if (m === 12) {
   386→    if (d >= 26 && d <= 31) return `${ordinal(d - 24)} jour de l’octave de Noël`
   387→  }
   388→  if (m === 1) {
   389→    if (d >= 2 && d <= 6) return `${ordinal(d)} jour avant l’Épiphanie`
   390→    if (d === 1) return 'Sainte Marie, Mère de Dieu'
   391→  }
   392→  return 'Temps de Noël'
   393→}

   395→function parseISO(iso: string): Date {
   396→  const [y, m, d] = iso.split('-').map(Number)
   397→  return new Date(y, (m || 1) - 1, d || 1)
   398→}
   399→
   400→function toISO(d: Date): string {
   401→  const y = d.getFullYear()
   402→  const m = String(d.getMonth() + 1).padStart(2, '0')
   403→  const day = String(d.getDate()).padStart(2, '0')
   404→  return `${y}-${m}-${day}`
   405→}

   407→export function getDayInfo(dateInput: string): DayInfo {
   408→  const date = parseISO(dateInput)
   409→  const season = determineSeason(date)
   410→  const isSunday = date.getDay() === 0
   411→  const feast = findFixedFeast(date)
   412→
   413→  // Determine the "principal celebration" of the day: Sundays and solemnities
   414→  // outrank fixed memorials; memorials are observed on weekdays.
   415→  let color: LiturgicalColor = season.color
   416→  let colorFilled: boolean
   417→  let dayName: string = season.dayName ?? ''
   418→  let feastLabel: string | undefined
   419→
   420→  // "Top-rank" days computed by the season — Sundays of Advent/Lent/Easter,
   421→  // Holy Week, Octave of Easter, Christmas octave, Trinity Sunday, Christ the
   422→  // King, Ash Wednesday, Holy Thursday, Good Friday, Holy Saturday, Ascension,
   423→  // Nativité, Baptême du Seigneur, Sainte Famille. These outrank ALL fixed
   424→  // feasts and memorials.
   425→  const seasonDayName = season.dayName ?? ''
   426→  const seasonIsTopRank =
   427→    (isSunday &&
   428→      (season.time === "Temps de l'Avent" ||
   429→        season.time === 'Temps du Carême' ||
   430→        season.time === 'Temps pascal' ||
   431→        season.time === 'Triduum pascal' ||
   432→        season.time === 'Temps de Noël')) ||
   433→    [
   434→      'Sainte Trinité',
   435→      'Christ Roi de l\'univers',
   436→      'Dimanche de Pâques',
   437→      'Dimanche de la Pentecôte',
   438→      'Dimanche des Rameaux et de la Passion du Seigneur',
   439→      'Mercredi des Cendres',
   440→      'Jeudi Saint',
   441→      'Vendredi Saint',
   442→      'Samedi Saint',
   443→      'Ascension du Seigneur',
   444→      'Nativité du Seigneur',
   445→      'Baptême du Seigneur',
   446→      'Sainte Famille',
   447→    ].includes(seasonDayName)
   448→
   449→  // Sundays of Ordinary Time are "solemnities of the Lord" (rank 3) — they
   450→  // outrank feasts and memorials, but are replaced by saints' solemnities.
   451→  const seasonIsSundayOT = isSunday && season.time === 'Temps ordinaire'
   452→
   453→  if (feast) {
   454→    const isSolemnity = feast.rank === 'solemnite'
   455→    const isFeast = feast.rank === 'fete'
   456→    if (seasonIsTopRank) {
   457→      // Season's top-rank day wins; fixed feast is noted (transferred/suppressed).
   458→      color = season.color
   459→      colorFilled = true
   460→      dayName = seasonDayName
   461→      feastLabel = feast.name
   462→    } else if (isSolemnity) {
   463→      // Fixed solemnity replaces the day (incl. Sunday of OT and weekdays).
   464→      color = feast.color
   465→      colorFilled = true
   466→      dayName = feast.name.replace(/ — solennité$/, '')
   467→      feastLabel = feast.name
   468→    } else if (seasonIsSundayOT) {
   469→      // Sunday of Ordinary Time wins over fixed feasts and memorials.
   470→      color = season.color
   471→      colorFilled = true
   472→      dayName = seasonDayName
   473→      feastLabel = feast.name
   474→    } else if (isFeast) {
   475→      // Feast wins for non-solemn weekdays.
   476→      color = feast.color
   477→      colorFilled = true
   478→      dayName = feast.name.replace(/ — fête$/, '')
   479→      feastLabel = feast.name
   480→    } else {
   481→      // Memorial on a weekday — memorial wins.
   482→      color = feast.color
   483→      colorFilled = feast.rank === 'memoire'
   484→      dayName = feast.name.replace(/ — mémoire$/, '').replace(/ — mémoire facultative$/, '')
   485→      feastLabel = feast.name
   486→    }
   487→  } else {
   488→    // No fixed feast.
   489→    if (isSunday) {
   490→      colorFilled = true // Sundays are always filled.
   491→    } else if (season.weekNumber > 0 && (season.time === 'Temps ordinaire' || season.time === 'Temps de l\'Avent' || season.time === 'Temps du Carême' || season.time === 'Temps pascal')) {
   492→      // Weekday of a liturgical season — outline badge.
   493→      colorFilled = false
   494→    } else {
   495→      colorFilled = false
   496→    }
   497→  }
   498→
   499→  // Sundays of Lent/Advent/Easter are filled.
   500→  if (isSunday && (season.time === 'Temps du Carême' || season.time === 'Temps de l\'Avent' || season.time === 'Temps pascal' || season.time === 'Triduum pascal')) {
   501→    colorFilled = true
   502→    color = season.color
   503→  }
   504→
   505→  if (!dayName) dayName = 'Férie'
   506→
   507→  return {
   508→    date: toISO(date),
   509→    color,
   510→    colorFilled,
   511→    liturgicalTime: season.time,
   512→    psalterWeek: psalterWeekFor(date),
   513→    year: `Année ${liturgicalYearLabel(date)}`,
   514→    dayName,
   515→    feast: feastLabel,
   516→  }
   517→}

   519→export function getMonthCalendar(year: number, month: number): DayInfo[] {
   520→  // month is 1-12.
   521→  const daysInMonth = new Date(year, month, 0).getDate()
   522→  const out: DayInfo[] = []
   523→  for (let d = 1; d <= daysInMonth; d++) {
   524→    const iso = `${year}-${String(month).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   525→    out.push(getDayInfo(iso))
   526→  }
   527→  return out
   528→}
