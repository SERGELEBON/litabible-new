   912→function interpolate(template: string, vars?: Record<string, string | number>): string {
   913→  if (!vars) return template
   914→  return template.replace(/\{(\w+)\}/g, (_m, k: string) => {
   915→    const v = vars[k]
   916→    return v === undefined ? `{${k}}` : String(v)
   917→  })
   918→}
   919→
   920→export function translate(locale: Locale, key: string, vars?: Record<string, string | number>): string {
   921→  const v = translations[locale]?.[key] ?? translations.en[key] ?? key
   922→  return interpolate(v, vars)
   923→}

   925→/**
   926→ * React hook returning a `t(key, vars?)` function bound to the current store
   927→ * locale, plus the `locale` itself.
   928→ *
   929→ * Missing keys fall back to English, then to the key itself — never crashes.
   930→ */
   931→export function useT() {
   932→  const locale = useLitabibleStore((s) => s.locale)
   933→  function t(key: string, vars?: Record<string, string | number>): string {
   934→    return translate(locale, key, vars)
   935→  }
   936→  return { t, locale }
   937→}

   969→// ---------------------------------------------------------------------------
   970→// Conclusion localizer — maps the French conclusion phrases to EN/ES.
   971→// Falls back to the original string when no match.

   973→
   974→const CONCLUSION_MAP: { fr: string; en: string; es: string }[] = [
   975→  { fr: '– Parole du Seigneur.', en: '– Word of the Lord.', es: '– Palabra del Señor.' },
   976→  { fr: '– Acclamons la Parole de Dieu.', en: '– Let us acclaim the Word of God.', es: '– Aclamemos la Palabra de Dios.' },
   977→  { fr: '– Louange à toi, Seigneur.', en: '– Praise to you, Lord.', es: '– Alabanza a ti, Señor.' },
   978→  { fr: '– Louange à toi, Seigneur Jésus.', en: '– Praise to you, Lord Jesus.', es: '– Alabanza a ti, Señor Jesús.' },
   979→]
   980→
   981→export function localizedConclusion(conclusion: string, locale: Locale): string {
   982→  if (locale === 'fr') return conclusion
   983→  const match = CONCLUSION_MAP.find((c) => c.fr === conclusion)
   984→  return match ? match[locale] : conclusion
   985→}
